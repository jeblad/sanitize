cmake_minimum_required(VERSION 3.19)

# Read version from package.json
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package.json" PACKAGE_JSON_CONTENT)
string(JSON PROJECT_VERSION_FROM_PACKAGE_JSON GET "${PACKAGE_JSON_CONTENT}" "version")

project(sanitize VERSION ${PROJECT_VERSION_FROM_PACKAGE_JSON})

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(SANITIZE_FULL_VERSION "v${PROJECT_VERSION}")
set(SANITIZE_VERSION_MAJOR ${PROJECT_VERSION_MAJOR})
set(SANITIZE_VERSION_MINOR ${PROJECT_VERSION_MINOR})
set(SANITIZE_VERSION_PATCH ${PROJECT_VERSION_PATCH})

# Written here rather than hardcoded in sanitize.hpp.in, so the .in file
# itself never displays a "do not edit" banner — it's the file meant to be edited.
set(SANITIZE_GENERATED_WARNING [[ * WARNING: DO NOT EDIT THIS FILE DIRECTLY.
 * This file is generated from sanitize.hpp.in via CMake.]])

# Generate the header directly into the source include directory — it is the
# "master" for distribution (header-only), tracked in git, regenerated here
# on every configure so it stays in sync with the version in package.json.
configure_file(sanitize.hpp.in "${CMAKE_CURRENT_SOURCE_DIR}/include/sanitize/sanitize.hpp" @ONLY)

add_library(sanitize INTERFACE)
target_include_directories(sanitize INTERFACE include)
target_compile_features(sanitize INTERFACE cxx_std_20) # Explicitly set C++20 features

# Enkel test-executable
add_executable(sanitize_test tests/test_sanitize.cpp)
target_link_libraries(sanitize_test sanitize)

enable_testing() # Enable CTest support
add_test(NAME sanitize_unit_tests COMMAND sanitize_test)
